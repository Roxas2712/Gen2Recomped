cmake_minimum_required(VERSION 3.20)
project(ascendant_tcg_gbc LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_STANDARD 17)
if(NOT GBRECOMP_ROOT OR NOT TCG_GENERATED)
  message(FATAL_ERROR "Set GBRECOMP_ROOT and TCG_GENERATED")
endif()
# The host supplies video, audio and input through LÖVE. No SDL window,
# runtime launcher or firmware file is needed for this explicit HLE build.
set(GBRT_BOOT_IMPLEMENTATION HLE CACHE STRING "Post-boot initialization" FORCE)
set(GBRECOMP_RECOMP_UI OFF CACHE BOOL "LÖVE owns the launcher" FORCE)
set(CMAKE_DISABLE_FIND_PACKAGE_SDL2 TRUE)
add_subdirectory("${GBRECOMP_ROOT}/runtime" runtime)
get_target_property(runtime_sources gbrt SOURCES)
list(FILTER runtime_sources EXCLUDE REGEX "(shader_pipeline.cpp|keybinds.c|platform_sdl.cpp|/gbrt.c|^src/gbrt.c)$")
set_property(TARGET gbrt PROPERTY SOURCES "${runtime_sources}")
# The pinned runtime's no-SDL stubs include the SDL key binding header
# unconditionally. Compile a private corrected copy; preserve upstream bytes.
file(READ "${GBRECOMP_ROOT}/runtime/src/platform_sdl.cpp" headless_platform)
string(REPLACE "#include \"keybinds.h\"" "#ifdef GB_HAS_SDL2\n#include \"keybinds.h\"\n#endif" headless_platform "${headless_platform}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/headless_platform.cpp" "${headless_platform}")
target_sources(gbrt PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/headless_platform.cpp")
target_sources(gbrt PRIVATE "${GBRECOMP_ROOT}/runtime/src/gbrt.c")
file(GLOB guest_sources CONFIGURE_DEPENDS "${TCG_GENERATED}/*.c")
list(FILTER guest_sources EXCLUDE REGEX "_main.c$")
if(NOT guest_sources)
  message(FATAL_ERROR "Generate the verified cartridge first")
endif()
add_library(ascendant_tcg_gbc SHARED bridge.cpp ${guest_sources})
target_include_directories(ascendant_tcg_gbc PRIVATE "${TCG_GENERATED}")
target_link_libraries(ascendant_tcg_gbc PRIVATE gbrt)

option(TCG_QA "Private bounded cartridge-routine qualification hooks" OFF)
if(TCG_QA)
  set_source_files_properties(bridge.cpp PROPERTIES COMPILE_DEFINITIONS ASCENDANT_TCG_QA=1)
endif()
