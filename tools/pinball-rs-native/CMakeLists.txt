cmake_minimum_required(VERSION 3.20)
project(ascendant_pinball_rs LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT GBARECOMP_ROOT OR NOT PINBALL_GENERATED OR NOT PINBALL_BIOS_GENERATED)
  message(FATAL_ERROR "Set GBARECOMP_ROOT, PINBALL_GENERATED and PINBALL_BIOS_GENERATED to the reviewed local inputs")
endif()
set(GBARECOMP_GENERATED_BIOS_DIR "${PINBALL_BIOS_GENERATED}" CACHE PATH "Local BIOS code" FORCE)
set(GBARECOMP_SELFHEAL_RECOMPILE_DEFAULT OFF CACHE BOOL "Explicitly opt in to runtime compilation" FORCE)
add_subdirectory("${GBARECOMP_ROOT}" framework EXCLUDE_FROM_ALL)
file(GLOB PINBALL_CODE CONFIGURE_DEPENDS "${PINBALL_GENERATED}/recompiled_*.cpp")
if(NOT PINBALL_CODE)
  message(FATAL_ERROR "Recompile the player's verified USA ROM first")
endif()
add_library(ascendant_pinball_rs SHARED bridge.cpp presentation.cpp ${PINBALL_CODE} "${PINBALL_GENERATED}/dispatch_table.cpp")
if(PINBALL_EU_VARIANT)
  target_compile_definitions(ascendant_pinball_rs PRIVATE PINBALL_EU_VARIANT=1)
  set_target_properties(ascendant_pinball_rs PROPERTIES OUTPUT_NAME ascendant_pinball_rs_eu)
endif()
target_include_directories(ascendant_pinball_rs PRIVATE "${PINBALL_GENERATED}" "${GBARECOMP_ROOT}/src/runtime" "${GBARECOMP_ROOT}/src/debug" "${GBARECOMP_ROOT}/src/gba")
target_link_gbarecomp_runtime_stack(ascendant_pinball_rs)
if(APPLE)
  target_link_options(ascendant_pinball_rs PRIVATE "-Wl,-rpath,@loader_path")
endif()
# Opt-in source tests; excluded from packaged builder inputs and normal builds.
option(PINBALL_PRESENTATION_TESTS "Build synthetic Pinball raster checks" OFF)
if(PINBALL_PRESENTATION_TESTS)
  add_executable(pinball_hud_test ../../tests/native/pinball_hud_test.cpp presentation.cpp)
  target_include_directories(pinball_hud_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${GBARECOMP_ROOT}/src/gba")
  target_link_gbarecomp_runtime_stack(pinball_hud_test)
  add_executable(pinball_obj_test ../../tests/native/pinball_obj_test.cpp)
  target_include_directories(pinball_obj_test PRIVATE "${GBARECOMP_ROOT}/src/gba")
  target_link_gbarecomp_runtime_stack(pinball_obj_test)
endif()
