cmake_minimum_required(VERSION 3.20)
project(ascendant_pinball_gbc LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_STANDARD 17)
if(NOT GBRECOMP_ROOT OR NOT PINBALL_GENERATED)
  message(FATAL_ERROR "Set GBRECOMP_ROOT and PINBALL_GENERATED")
endif()
# The host supplies video, audio and input through LÖVE. No SDL window,
# runtime launcher or firmware file is needed for this explicit HLE build.
set(GBRT_BOOT_IMPLEMENTATION HLE CACHE STRING "Post-boot initialization" FORCE)
set(GBRECOMP_RECOMP_UI OFF CACHE BOOL "LÖVE owns the launcher" FORCE)
set(CMAKE_DISABLE_FIND_PACKAGE_SDL2 TRUE)
add_subdirectory("${GBRECOMP_ROOT}/runtime" runtime)
get_target_property(runtime_sources gbrt SOURCES)
list(FILTER runtime_sources EXCLUDE REGEX "(shader_pipeline.cpp|keybinds.c|platform_sdl.cpp|/gbrt.c|^src/gbrt.c)$")
set_property(TARGET gbrt PROPERTY SOURCES "${runtime_sources}")
# The pinned runtime's no-SDL stubs include the SDL key binding header
# unconditionally. Compile a private corrected copy; preserve upstream bytes.
file(READ "${GBRECOMP_ROOT}/runtime/src/platform_sdl.cpp" headless_platform)
string(REPLACE "#include \"keybinds.h\"" "#ifdef GB_HAS_SDL2\n#include \"keybinds.h\"\n#endif" headless_platform "${headless_platform}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/headless_platform.cpp" "${headless_platform}")
target_sources(gbrt PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/headless_platform.cpp")
# On rumble cartridges bit 3 drives the motor, not a fourth RAM-bank bit
# (Pan Docs, MBC5). Keep this correction local to the private build.
file(READ "${GBRECOMP_ROOT}/runtime/src/gbrt.c" gbc_runtime)
string(FIND "${gbc_runtime}" "ctx->ram_bank = value & 0x0F;" mbc5_site)
if(mbc5_site EQUAL -1)
  message(FATAL_ERROR "Pinned MBC5 correction site changed; review the runtime")
endif()
string(REPLACE "ctx->ram_bank = value & 0x0F;" "ctx->ram_bank = value & (ctx->mbc_type >= 0x1C ? 0x07 : 0x0F);\n                pinball_gbc_mbc5_rumble(ctx, value);" gbc_runtime "${gbc_runtime}")
string(REPLACE "#include \"gbrt.h\"" "#include \"gbrt.h\"\nextern void pinball_gbc_mbc5_rumble(GBContext*, uint8_t);" gbc_runtime "${gbc_runtime}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/gbc_runtime.c" "${gbc_runtime}")
target_sources(gbrt PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/gbc_runtime.c")
target_include_directories(gbrt PRIVATE "${GBRECOMP_ROOT}/runtime/src")
file(GLOB guest_sources CONFIGURE_DEPENDS "${PINBALL_GENERATED}/*.c")
list(FILTER guest_sources EXCLUDE REGEX "_main.c$")
if(NOT guest_sources)
  message(FATAL_ERROR "Generate the verified cartridge first")
endif()
add_library(ascendant_pinball_gbc SHARED bridge.cpp table.cpp ${guest_sources})
target_include_directories(ascendant_pinball_gbc PRIVATE "${PINBALL_GENERATED}")
target_link_libraries(ascendant_pinball_gbc PRIVATE gbrt)
