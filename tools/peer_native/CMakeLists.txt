cmake_minimum_required(VERSION 3.21)
project(retro_peer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(NO_MEDIA ON CACHE BOOL "" FORCE)
set(NO_EXAMPLES ON CACHE BOOL "" FORCE)
set(NO_TESTS ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
if(NOT DATACHANNEL_DIR)
  message(FATAL_ERROR "Provide the pinned libdatachannel v0.24.6 source directory")
endif()
add_subdirectory(${DATACHANNEL_DIR} datachannel)
add_library(retro_peer SHARED bridge.cpp)
target_link_libraries(retro_peer PRIVATE LibDataChannel::LibDataChannelStatic)
set_target_properties(retro_peer PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
